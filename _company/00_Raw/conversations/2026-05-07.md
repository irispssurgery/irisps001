### 🛠️ FFmpeg 렌더링 자동 검증 스크립트 (`render_validator.py`)

이 스크립트는 지정된 파일 리스트를 전수 조사하여, 하나라도 누락되면 즉시 중단하고 누락된 파일명을 출력합니다. 모든 파일이 존재할 경우에만 FFmpeg 명령어를 실행합니다.

```python
import os
import subprocess
import sys

# 1. 검증이 필요한 소스 파일 리스트 (사용자 환경에 맞게 수정)
REQUIRED_FILES = [
    "input_video.mp4",
    "background_audio.mp3",
    "overlay_image.png",
    "subtitles.srt"
]

# 2. 실행할 FFmpeg 명령어 (리스트 형태로 작성)
# 예시: 영상 + 오디오 + 이미지 합성 명령어
FFMPEG_COMMAND = [
    "ffmpeg",
    "-y",                # 기존 파일 덮어쓰기 허용
    "-i", "input_video.mp4",
    "-i", "background_audio.mp3",
    "-i", "overlay_image.png",
    "-filter_complex", "[0:v][2:v]overlay=10:10[outv]",
    "-map", "[outv]",
    "-map", "1:a",
    "-c:v", "libx264",
    "-c:a", "aac",
    "-shortest",
    "final_output.mp4"
]

def check_files(file_list):
    """파일 존재 여부를 확인하고 누락된 파일 목록을 반환합니다."""
    missing_files = [f for f in file_list if not os.path.exists(f)]
    return missing_files

def run_render(command):
    """FFmpeg 명령어를 실행합니다."""
    print(f"🚀 Rendering started: {' '.join(command)}")
    try:
        # check=True: 명령어 실패 시 CalledProcessError 발생
        subprocess.run(command, check=True)
        print("✅ Rendering completed successfully!")
    except subprocess.CalledProcessError as e:
        print(f"❌ FFmpeg Error: Command failed with return code {e.returncode}")
        sys.exit(1)
    except FileNotFoundError:
        print("❌ Error: FFmpeg is not installed or not in PATH.")
        sys.exit(1)

if __name__ == "__main__":
    print("🔍 Checking source files...")
    missing = check_files(REQUIRED_FILES)

    if missing:
        print("❌ Error: Missing required source files:")
        for m in missing:
            print(f"  - {m}")
        print("\n⚠️ Please ensure all files are present before re-running.")
        sys.exit(1)
    else:
        print("✅ All source files found.")
        run_render(FFMPEG_COMMAND)
```

### 🚀 사용 방법
1. `REQUIRED_FILES` 리스트에 누락되었던 파일 경로들을 모두 입력합니다.
2. `FFMPEG_COMMAND`에 실행하고자 하는 최종 명령어를 리스트 형태로 입력합니다.
3. 터미널에서 아래 명령어를 실행합니다.
   ```bash
   python render_validator.py
   ```

```python
import os
import subprocess
import sys

def safe_render(input_files, output_file):
    """
    모든 입력 경로를 정규화하고 따옴표로 감싸 FFmpeg 실행 오류를 방지함.
    """
    # 1. Pre-Verification: 파일 존재 여부 및 경로 정규화
    validated_inputs = []
    for f in input_files:
        abs_path = os.path.abspath(f)
        if not os.path.exists(abs_path):
            print(f"❌ [Error] 소스 파일 누락: {abs_path}")
            return False
        # 경로에 공백이 있어도 안전하도록 따옴표 처리된 형태로 리스트에 저장
        validated_inputs.append(abs_path)

    # 2. Command Construction: FFmpeg 명령어 생성 (인자별 따옴표 처리)
    # FFmpeg는 인자 자체에 따옴표가 포함된 문자열을 받을 때 내부적으로 처리하므로, 
    # subprocess 리스트 인자 전달 시 경로를 그대로 넣되, 시스템 호출 시의 안전성을 확보함.
    cmd = ["ffmpeg", "-y"]
    for inp in validated_inputs:
        cmd.extend(["-i", inp])
    
    # 예시: 단순 이미지/오디오 병합 (실제 프로젝트에 맞게 필터 수정 필요)
    # 여기서는 렌더링 성공 여부 확인을 위한 최소 구조만 작성
    cmd.extend(["-c:v", "libx264", "-pix_fmt", "yuv420p", output_file])

    print(f"🚀 실행 명령어: {' '.join(['\"' + x + '\"' for x in cmd])}")

    # 3. Execution
    try:
        result = subprocess.run(cmd, check=True, capture_output=True, text=True)
        print("✅ [Success] 렌더링 완료.")
        return True
    except subprocess.CalledProcessError as e:
        print(f"❌ [Failure] FFmpeg 에러 발생 (Exit Code: {e.returncode})")
        print(f" stderr: {e.stderr}")
